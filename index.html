SELECT
    customer_id,
    COUNT(*) AS total_orders,
    SUM(amount) AS total_spent
FROM orders
WHERE order_date >= '2025-01-01'
GROUP BY customer_id
ORDER BY total_spent DESC
LIMIT 10;

    date,

incremental_load.pypython
def incremental_load(new_data, existing_table, timestamp_column='updated_at'):
    # Latest timestamp already loaded
    latest = existing_table[timestamp_column].max()
    # Keep only new or updated records
    new_records = new_data[new_data[timestamp_column] > latest]
    return pd.concat([existing_table, new_records], ignore_index=True)
Process only changed records.

-- Group by month, quarter, or year
SELECT
    DATE_TRUNC('month', transaction_date) AS month,
    SUM(amount) AS monthly_revenue
FROM transactions
GROUP BY 1
ORDER BY 1 DESC;

left_join_coalesce.sqlsql
SELECT
    u.user_id,
    u.name,
    COALESCE(o.total_orders, 0) AS order_count,
    COALESCE(o.total_spent, 0) AS lifetime_value
FROM users u
LEFT JOIN (
    SELECT user_id, COUNT(*) AS total_orders, SUM(amount) AS total_spent
    FROM orders
    GROUP BY user_id
) o ON u.user_id = o.user_id;
Keep all users; replace NULL with 0.
